% Remove duplicates
remove_duplicate([], []).

remove_duplicate([Head|Tail], Listnew) :- member(Head, Tail), remove_duplicate(Tail, Listnew).

remove_duplicate([Head|Tail], [Head|Listnew]) :- \+ member(Head, Tail), remove_duplicate(Tail, Listnew).

% Extract prerequisites
get_all_prerequisites(Course, Prerequisites) :- findall(One_Answer, prereq(Course, One_Answer), Next), remove_duplicate(Next, Prerequisites).

prereq(Prerequisites, One_Answer) :- prerequisite(Prerequisites, One_Answer).

prereq(Prerequisites, Next) :- prerequisite(Prerequisites, One_Answer), prereq(One_Answer, Next).
